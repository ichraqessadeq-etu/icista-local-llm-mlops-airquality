```yaml
name: CI/CD Pipeline

on: [push, pull_request]

  build:
    name: Build and Test
    runs-on: ubuntu-latest

    steps:
    - uses: actions/checkout@v2

    - name: Setup Python
      uses: actions/setup-python@v2
      with:
        python-version: 3.8

    - name: Install Dependencies
      run: |
        pip install pandas numpy mlflow pytest

    - name: Test
      run: pytest test_pipeline.py

  train:
    name: Train Model
    needs: build
    runs-on: ubuntu-latest

    steps:
    - name: Load Data
      run: |
        data=$(cat data/air_quality.csv | python pipeline.py load_data)

    - name: Preprocess Data
      run: |
        X, y = python pipeline.py preprocess_data $data

    - name: Train Model
      run: |
        model=$(python pipeline.py train_model $X $y)
        echo "Model Trained Successfully"

    - name: Log Model
      run: |
        mlflow login --host $MLFLOW_TRACKING_URI --username $MLFLOW_USER --password $MLFLOW_PASSWORD
        python pipeline.py log_model $model
```

This setup uses MLflow for model training and logging, pytest for unit testing, and GitHub Actions for CI/CD. You'll need to replace `$MLFLOW_TRACKING_URI`, `$MLFLOW_USER`, and `$MLFLOW_PASSWORD` with your actual MLflow tracking server details.
