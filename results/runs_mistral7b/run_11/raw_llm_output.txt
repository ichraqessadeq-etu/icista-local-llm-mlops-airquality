3. **GitHub Actions YAML**

```yaml
name: CI/CD Pipeline

on: [push, pull_request]

jobs:
  build:
    name: Build and Test
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v2
      - name: Setup Python
        uses: actions/setup-python@v2
        with:
          python-version: 3.8
      - name: Install dependencies
        run: |
          pip install pandas mlflow pytest
      - name: Test
        run: pytest test_pipeline.py
      - name: Train Model
        run: python pipeline.py data/air_quality.csv

  deploy:
    name: Deploy Model
    needs: build
    runs-on: ubuntu-latest
    steps:
      - name: Save artifacts
        uses: actions/upload-artifact@v2
        with:
          name: model
          path: mlruns/$(mlflow_run_id)
      - name: Deploy Model
        uses: mlflow/mlflow-action@v2
        with:
          track-run: true
          artifact: model
```

This pipeline uses MLflow for model training, logging, and deployment. It includes unit tests for the Python functions and GitHub Actions for continuous integration and deployment. The CI workflow runs tests on every push or pull request, while the CD workflow deploys the model after a successful build.
